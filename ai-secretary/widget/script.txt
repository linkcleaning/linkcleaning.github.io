const DATA = "https://linkcleaning.github.io/ai-secretary/data/today.json";
const APP = "https://linkcleaning.github.io/ai-secretary/";
const YELLOW = new Color("#FFD233");
const INK = new Color("#191919");
const MUTED = new Color("#45474D");
function ymd(d) { const p = function (n) { return String(n).padStart(2, "0"); }; return d.getFullYear() + "-" + p(d.getMonth() + 1) + "-" + p(d.getDate()); }
let data = null;
try { const r = new Request(DATA + "?t=" + Date.now()); r.timeoutInterval = 10; data = await r.loadJSON(); } catch (e) { data = null; }
const fam = config.widgetFamily || "medium";
const w = new ListWidget();
w.backgroundColor = YELLOW;
w.url = APP;
w.setPadding(14, 16, 14, 16);
w.refreshAfterDate = new Date(Date.now() + 3600000);
const now = new Date();
const top = w.addStack();
top.centerAlignContent();
const badge = top.addStack();
badge.backgroundColor = INK;
badge.cornerRadius = 9;
badge.setPadding(2, 7, 2, 7);
const bt = badge.addText("미래전략실");
bt.font = Font.heavySystemFont(10);
bt.textColor = YELLOW;
top.addSpacer(6);
const dt = top.addText((now.getMonth() + 1) + "." + now.getDate() + " (" + "일월화수목금토"[now.getDay()] + ")");
dt.font = Font.heavySystemFont(12);
dt.textColor = INK;
top.addSpacer();
const ver = top.addText("v2");
ver.font = Font.systemFont(7);
ver.textColor = MUTED;
w.addSpacer(8);
const today = ymd(now);
const tomorrow = ymd(new Date(now.getTime() + 86400000));
if (!data || (data.date !== today && data.date !== tomorrow)) {
  const t = w.addText("오늘 업무는 아침 7시 17분에 준비돼요");
  t.font = Font.boldSystemFont(14);
  t.textColor = INK;
} else {
  if (fam !== "small" && data.headline) {
    const h = w.addText(String(data.headline).replace(/^대표님,\s*/, ""));
    h.font = Font.semiboldSystemFont(fam === "large" ? 14 : 12);
    h.textColor = MUTED;
    h.lineLimit = fam === "large" ? 4 : 2;
    w.addSpacer(8);
  }
  const label = w.addText(data.date === tomorrow ? "내일 할 일" : "오늘 할 일");
  label.font = Font.heavySystemFont(11);
  label.textColor = MUTED;
  w.addSpacer(3);
  const todo = (data.todo || []).slice(0, fam === "small" ? 2 : 3);
  for (let i = 0; i < todo.length; i++) {
    const row = w.addText((i + 1) + ". " + String(todo[i]).replace(/\s*\(.*?\)\s*$/, ""));
    row.font = Font.boldSystemFont(fam === "small" ? 12 : 13);
    row.textColor = INK;
    row.lineLimit = fam === "large" ? 2 : 1;
    w.addSpacer(2);
  }
}
w.addSpacer();
if (config.runsInWidget) { Script.setWidget(w); } else { await w.presentMedium(); }
Script.complete();
